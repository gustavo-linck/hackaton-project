# UmbLink Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` or `superpowers:executing-plans` to implement this plan task-by-task.

**Goal:** Construir o UmbLink — criador de páginas de links freemium com editor em tempo real, analytics, gestão de planos e painel admin, deployável via Docker.

**Architecture:** Clean Architecture 3 camadas (Web/Blazor Server, Application/Services, Infrastructure/EF Core). Limites de plano centralizados em `PlanLimitService`. Métricas assíncronas via `Channel<T>` + `BackgroundService`.

**Tech Stack:** .NET 8, Blazor Server (`@rendermode InteractiveServer`), EF Core + SQLite, ASP.NET Identity, Google OAuth2, FluentValidation, Bootstrap 5, SortableJS (drag-and-drop), Chart.js (gráficos)

**Spec:** `docs/superpowers/specs/2026-03-28-umblink-design.md`

```csharp
public class Result<T>
{
    public T? Value { get; }
    public string? Error { get; }
    public bool IsSuccess { get; }

    public LimitExceededError? LimitError { get; }

    private Result(T value) { Value = value; IsSuccess = true; }
    private Result(string error, LimitExceededError? limitError = null)

    public static Result<T> Ok(T value) => new(value);
```

```csharp
public class PlanLimitService(AppDbContext db) : IPlanLimitService
{
    public async Task<Result<bool>> CanAddPageAsync(Guid userId)
    {
        var limits = await GetLimitsAsync(userId);
        if (limits.MaxPages == -1) return Result<bool>.Ok(true);

        var count = await db.Pages.CountAsync(p => p.UserId == userId
            && p.Status != PageStatus.Suspended);
        if (count >= limits.MaxPages)
            return Result<bool>.LimitExceeded(new LimitExceededError
            {
                FeatureName = "Páginas",
                Message = $"Você atingiu o limite de {limits.MaxPages} página(s) do plano atual.",
                CurrentPlan = await GetPlanNameAsync(userId),
                RequiredPlan = limits.MaxPages == 1 ? "Pro" : "Business"
            });
        return Result<bool>.Ok(true);
    }

    public async Task<Result<bool>> CanAddLinkAsync(Guid userId, Guid pageId)
    {
        var limits = await GetLimitsAsync(userId);
        if (limits.MaxLinksPerPage == -1) return Result<bool>.Ok(true);

        var count = await db.Links.CountAsync(l => l.PageId == pageId);
        if (count >= limits.MaxLinksPerPage)
            return Result<bool>.LimitExceeded(new LimitExceededError
            {
                FeatureName = "Links",
                Message = $"Limite de {limits.MaxLinksPerPage} links por página atingido.",
                CurrentPlan = await GetPlanNameAsync(userId),
                RequiredPlan = limits.MaxLinksPerPage == 3 ? "Pro" : "Business"
            });
        return Result<bool>.Ok(true);
    }

    public async Task<Result<bool>> CanUseFeatureAsync(Guid userId, Feature feature)
    {
        var limits = await GetLimitsAsync(userId);
        var (allowed, requiredPlan, name) = feature switch
        {
            Feature.CustomDomain     => (limits.AllowCustomDomain, "Pro", "Domínio customizado"),
            Feature.RemoveBranding   => (limits.AllowRemoveBranding, "Business", "Remoção de branding"),
            Feature.Referrer         => (limits.AllowReferrer, "Pro", "Origem de acesso"),
            Feature.AdvancedThemes   => (limits.ThemeCount == -1 || limits.ThemeCount > 2, "Pro", "Temas avançados"),
            Feature.AdvancedFonts    => (limits.FontCount == -1 || limits.FontCount > 2, "Pro", "Fontes avançadas"),
            Feature.AdvancedColors   => (limits.AllowCustomDomain, "Pro", "Personalização de cores"), // proxy: Pro+
            _ => (false, "Pro", feature.ToString())
        };

        if (!allowed)
            return Result<bool>.LimitExceeded(new LimitExceededError
            {
                FeatureName = name,
                Message = $"{name} está disponível a partir do plano {requiredPlan}.",
                CurrentPlan = await GetPlanNameAsync(userId),
                RequiredPlan = requiredPlan
            });
        return Result<bool>.Ok(true);
    }

    public async Task<PlanLimit> GetLimitsAsync(Guid userId)
    {
        var planId = await db.Subscriptions
            .Where(s => s.UserId == userId)
            .Select(s => s.PlanId)
            .FirstOrDefaultAsync();
        return await db.PlanLimits.FirstAsync(pl => pl.PlanId == planId);
    }

    private async Task<string> GetPlanNameAsync(Guid userId) =>
        await db.Subscriptions.Where(s => s.UserId == userId)
            .Select(s => s.Plan.Name).FirstOrDefaultAsync() ?? "Free";
}
```

```csharp
public async Task<MetricsSummaryDto> GetPageMetricsAsync(Guid userId, Guid pageId, int days)
{
    using var scope = _sp.CreateScope();
    var db = scope.ServiceProvider.GetRequiredService<AppDbContext>();

    var cutoff = days == -1 ? DateTime.MinValue : DateTime.UtcNow.AddDays(-days);

    var views = await db.PageViews
        .Where(v => v.PageId == pageId && v.Timestamp >= cutoff)
        .GroupBy(v => v.Timestamp.Date)
        .Select(g => new DailyMetricDto(g.Key, g.Count(), 0))
        .ToListAsync();

    var links = await db.Links.Where(l => l.PageId == pageId).ToListAsync();
    var totalViews = views.Sum(v => v.Views);

    var linkMetrics = new List<LinkMetricDto>();
    foreach (var link in links)
    {
        var clicks = await db.ClickEvents
            .CountAsync(c => c.LinkId == link.Id && c.Timestamp >= cutoff);
        linkMetrics.Add(new LinkMetricDto(link.Id, link.Title, clicks,
            totalViews > 0 ? Math.Round((double)clicks / totalViews * 100, 1) : 0));
    }

    return new MetricsSummaryDto(totalViews, linkMetrics.Sum(l => l.Clicks),
        views, linkMetrics.OrderByDescending(l => l.Clicks).ToList());
}
```
